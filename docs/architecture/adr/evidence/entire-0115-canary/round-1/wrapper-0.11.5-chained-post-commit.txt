#!/bin/sh
# Entire CLI hooks
# Post-commit hook: condense session data if commit has Entire-Checkpoint trailer
if command -v entire >/dev/null 2>&1; then entire hooks git post-commit 2>/dev/null || true; else :; fi
# Chain: run pre-existing hook
_entire_hook_dir="$(dirname "$0")"
if [ -z "${ENTIRE_CHAINING_POST_COMMIT:-}" ]; then
    ENTIRE_CHAINING_POST_COMMIT=1
    export ENTIRE_CHAINING_POST_COMMIT
    if [ -n "${PRE_COMMIT_RUNNING_LEGACY:-}" ] && grep -q '^# File generated by pre-commit: https://pre-commit.com' "$_entire_hook_dir/post-commit.pre-entire" 2>/dev/null; then
        if [ -x "$_entire_hook_dir/post-commit.legacy.pre-entire" ]; then
            "$_entire_hook_dir/post-commit.legacy.pre-entire" "$@"
        fi
    elif [ -x "$_entire_hook_dir/post-commit.pre-entire" ]; then
        "$_entire_hook_dir/post-commit.pre-entire" "$@"
    fi
fi
