#!/bin/sh
# Entire CLI hooks
# Post-rewrite hook: remap session linkage after amend/rebase rewrites
if command -v entire >/dev/null 2>&1; then sh -c 'g="$(git rev-parse --show-toplevel)/scripts/entire/capture_guard.sh"; if [ -x "$g" ]; then exec "$g" "$@"; fi; echo "[entire] capture refused (guard_missing): no checkpoint written" >&2' entire-capture-guard hooks git post-rewrite "$1" 2>/dev/null || true; else :; fi
